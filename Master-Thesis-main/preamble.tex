% ============================================================
% REGISTRO DEI REVISORI (seconda colonna del margine)
%
% Le note di MD restano dove sono: macro \MD*, todonotes, colonna di SINISTRA,

% Aggiungere un revisore richiede una riga sola:

% \leavevmode: senza, una nota messa subito dopo un titolo o una \label parte
% in modo verticale, il \marginpar non puo' migrare e LaTeX perde i float
% della pagina ("Float(s) lost").
\newcommand{\revnotegen}[6]{% #6 = nop | nsv% #1 sigla  #2 colore  #3 categoria  #4 ancora  #5 nota
  \leavevmode
  \csname ifrevshow#1\endcsname
    \dashuline{#4}%
    \refstepcounter{rev#1}%
    \textsuperscript{\textbf{\scriptsize[#1\,\csname therev#1\endcsname]}}%
    \addcontentsline{cco}{todo}{%
      \protect\fcolorbox{black}{#2}{\protect\textcolor{#2}{o}}\ %
      \protect\textbf{[#1\,\csname therev#1\endcsname] #3:} #4}%
    \csname note#6entry\endcsname{#2}{[#1\,\csname therev#1\endcsname] #3:}{#5}%
    \marginpar{\makebox[0pt][l]{\hspace*{10.5cm}%
      \fcolorbox{black}{#2}{\parbox{11.6cm}{\scriptsize\raggedright
        \textbf{[#1\,\csname therev#1\endcsname] #3:} #5}}}}%
  \else #4\fi
}

% NOTE GENERALI
%
% Un terzo tipo di nota, accanto a quelle ancorate a una frase: riguarda un
% CAPITOLO o una SEZIONE INTERA, non un punto preciso del testo. Serve per le
% osservazioni che non hanno una frase a cui appendersi -- notazione non
% uniforme, una sezione scollegata dal resto, squilibri di profondita' o di
% bibliografia.
%
%   \CCgen{ambito}{titolo breve}{testo}
%
% Numerazione separata ([CC G1], [CC G2], ...), colore proprio, elenco proprio
% (\listofgeneralnotes). Entra comunque anche negli elenchi per stato.
%
% Va messa DENTRO il primo capoverso della sezione a cui si riferisce, mai
% subito dopo il titolo: un \marginpar aperto in modo verticale fa perdere i
% float della pagina.
\newcommand{\revgeneral}[5]{% #1 sigla  #2 colore  #3 ambito  #4 titolo  #5 testo
  \leavevmode
  \csname ifrevshow#1\endcsname
    \refstepcounter{gen#1}%
    \textsuperscript{\textbf{\scriptsize[#1\,G\csname thegen#1\endcsname]}}%
    \addcontentsline{gen}{todo}{%
      \protect\fcolorbox{black}{#2}{\protect\textcolor{#2}{o}}\ %
      \protect\textbf{[#1\,G\csname thegen#1\endcsname] #3 -- #4:} #5}%
    \noteopenentry{#2}{[#1\,G\csname thegen#1\endcsname] generale (#3) -- #4:}{#5}%
    \marginpar{\makebox[0pt][l]{\hspace*{10.5cm}%
      \fcolorbox{black}{#2}{\parbox{11.6cm}{\scriptsize\raggedright
        \textbf{[#1\,G\csname thegen#1\endcsname] nota generale, #3 -- #4:} #5}}}}%
  \fi
}
