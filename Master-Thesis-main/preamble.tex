% NOTE DI CLAUDE CODE (CC) -- seconda colonna, a destra di quella di MD.
%
% Le note di MD restano esattamente come sono: stesse macro, stesso colore,
% stessa colonna, e finiscono come prima in \listoftodos. Le note CC sono un
% livello separato, disegnato con \marginpar a 10.5 cm di distanza, e NON

%
% I marcatori nel testo sono deliberatamente diversi: MD evidenzia (soul) o usa
% l'onda (ulem), CC sottolinea tratteggiato. CC usa solo ulem, mai soul, e
% quindi non soffre del limite con wrapfig documentato sotto.
%
% Categorie: non un calco di quelle di MD. Quelle di MD dicono che tipo di
% intervento serve (nota, domanda, suggerimento, refuso); queste dicono che
% tipo di AFFERMAZIONE e', che in revisione e' l'informazione utile.
%   \CCerr  errore verificato NEL TESTO della tesi
%   \CCn    osservazione tecnica sul testo: cosa manca o cosa e' impreciso
%   \CCref  rilievo meccanico: bibliografia, figure, riferimenti incrociati
%   \CCmd   verdetto su una nota di MD (non evidenzia nulla: commenta la nota)
% ============================================================
% ============================================================
% REGISTRO DEI REVISORI (seconda colonna del margine)
%
% Le note di MD restano dove sono: macro \MD*, todonotes, colonna di SINISTRA,
% ed e' l'unico contenuto di \listoftodos. Tutti gli ALTRI revisori condividono
% la colonna di DESTRA e si distinguono per colore, sigla e numerazione propria.
%
% Aggiungere un revisore richiede una riga sola:
%     \DeclareReviewer{GS}{blue!15}
% e da quel momento esiste la macro \GS{categoria}{ancora}{nota}, per esempio
%     \GS{errore}{the measured waist agrees}{Il confronto non regge: ...}
% che stampa [GS 1] nel testo, mette il riquadro blu nella seconda colonna e
% aggiunge una riga all'elenco dei revisori.
%
% Il colore identifica la PERSONA, non la categoria: con piu' revisori e' il
% colore che deve dire subito di chi e' la nota. La categoria resta la parola
% in grassetto all'inizio del riquadro.
%
% \hidereviewer{GS} spegne un layer: le note di quel revisore spariscono da
% testo, margine ed elenco, e resta solo il testo della tesi. Serve per produrre
% un PDF per revisore senza toccare le note.
% ============================================================

\newcommand{\revnote}[5]{% #1 sigla  #2 colore  #3 categoria  #4 ancora  #5 nota
  \csname ifrevshow#1\endcsname
    \dashuline{#4}%
    \refstepcounter{rev#1}%
    \textsuperscript{\textbf{\scriptsize[#1\,\csname therev#1\endcsname]}}%
    \addcontentsline{cco}{todo}{%
      \protect\fcolorbox{black}{#2}{\protect\textcolor{#2}{o}}\ %
      \protect\textbf{[#1\,\csname therev#1\endcsname] #3:} #4}%
    \marginpar{\makebox[0pt][l]{\hspace*{10.5cm}%
      \fcolorbox{black}{#2}{\parbox{11.6cm}{\scriptsize\raggedright
        \textbf{[#1\,\csname therev#1\endcsname] #3:} #5}}}}%
  \else #4\fi
}
% variante senza ancoraggio nel testo: #4 e' solo l'etichetta per l'elenco
\newcommand{\revnoteplain}[5]{%
  \csname ifrevshow#1\endcsname
    \refstepcounter{rev#1}%
    \textsuperscript{\textbf{\scriptsize[#1\,\csname therev#1\endcsname]}}%
    \addcontentsline{cco}{todo}{%
      \protect\fcolorbox{black}{#2}{\protect\textcolor{#2}{o}}\ %
      \protect\textbf{[#1\,\csname therev#1\endcsname] #3:} #4}%
    \marginpar{\makebox[0pt][l]{\hspace*{10.5cm}%
      \fcolorbox{black}{#2}{\parbox{11.6cm}{\scriptsize\raggedright
        \textbf{[#1\,\csname therev#1\endcsname] #3:} #5}}}}%
  \fi
}

  \newcounter{rev#1}%

  \csname revshow#1true\endcsname

}
